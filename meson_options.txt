# Build options for Launch-Time testing tool (lttt) project

option('enable_debug', type: 'boolean', value: false,
       description: 'Enable debug build with extra logging')

option('enable_tests', type: 'boolean', value: false,
       description: 'Enable building of unit tests')

option('max_lcores', type: 'integer', value: 128, min: 1, max: 1024,
       description: 'Maximum number of logical cores supported')

option('enable_mqtt', type: 'boolean', value: true,
       description: 'Enable MQTT support for telemetry')
